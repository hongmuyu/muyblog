## 🛠️ 第二类：高效数据处理

### 6. 前缀和
```python
# 快速求区间和：空间换时间
pre = [0] * n
pre[0] = nums[0]
for i in range(1, n):
    pre[i] = pre[i-1] + nums[i]
# 区间 [l,r] 的和 = pre[r] - pre[l-1]
```

### 7. 差分
```python
# 区间加减最后还原：先记变化量
diff = [0] * (n+1)
for l, r, val in 操作列表:
    diff[l] += val
    diff[r+1] -= val
# 还原：前缀和
res = [0] * n
res[0] = diff[0]
for i in range(1, n):
    res[i] = res[i-1] + diff[i]
```

### 8. 离散化
```python
# 大值映射到小索引：排序 + 字典
vals = sorted(set(arr))
mp = {v:i for i,v in enumerate(vals)}
new_arr = [mp[x] for x in arr]
```

## 🚀 第三类：进阶策略

### 9. 贪心
```python
# 每一步选当前最优，不回头
# 典型：区间选点 – 按结束时间排序
intervals.sort(key=lambda x: x[1])
cnt, end = 0, -inf
for s, e in intervals:
    if s >= end:
        cnt += 1
        end = e
```

### 10. 双指针
```python
# 对撞指针（两数之和）
l, r = 0, len(arr)-1
while l < r:
    s = arr[l] + arr[r]
    if s == target: break
    elif s < target: l += 1
    else: r -= 1

# 快慢指针（链表判环）
slow = fast = head
while fast and fast.next:
    slow = slow.next
    fast = fast.next.next
    if slow == fast: 有环
```

### 13. 构造
```python
# 找规律，直接造出答案
# 例：1, n, 2, n-1, 3, ...
l, r = 1, n
res = []
while l <= r:
    res.append(l); l+=1
    if l<=r: res.append(r); r-=1
```

## 🌲 DFS 系列

### DFS 基础（n重循环替代）
```python
def dfs(depth):
    if depth == N:          # 最内层
        # 处理当前组合
        return
    for i in range(可选范围):
        # 做选择
        dfs(depth+1)
```

### 回溯（全排列模板）
```python
def backtrack(path, used):
    if len(path) == n:
        res.append(path[:])
        return
    for i in range(n):
        if used[i]: continue
        used[i] = True
        path.append(nums[i])
        backtrack(path, used)
        path.pop()          # 撤销
        used[i] = False
```

### 剪枝
```python
# 在回溯中加入：如果当前已经不可能比最优解好，直接 return
if current_cost >= best:
    return
# 或可行性剪枝：当前状态不可能合法
if not is_valid(state):
    return
```

### 记忆化搜索
```python
from functools import lru_cache

@lru_cache(None)
def dfs(state):
    if 终止: return 1
    ans = 0
    for nxt in 转移(state):
        ans += dfs(nxt)
    return ans
```

## 🧱 动态规划

### 线性DP（斐波那契）
```python
dp = [0] * (n+1)
dp[0], dp[1] = 0, 1
for i in range(2, n+1):
    dp[i] = dp[i-1] + dp[i-2]
```

### 0/1背包
```python
dp = [0] * (C+1)
for w, v in items:
    for j in range(C, w-1, -1):   # 倒序！保证每个物品只选一次
        dp[j] = max(dp[j], dp[j-w] + v)
```

### 完全背包
```python
# 正序遍历，物品可以选多次
for j in range(w, C+1):
    dp[j] = max(dp[j], dp[j-w] + v)
```

### 区间DP（石子合并）
```python
dp = [[0]*(n+1) for _ in range(n+1)]
for length in range(2, n+1):          # 区间长度
    for i in range(1, n-length+2):    # 起点
        j = i+length-1                 # 终点
        dp[i][j] = inf
        for k in range(i, j):          # 分割点
            dp[i][j] = min(dp[i][j], dp[i][k]+dp[k+1][j] + cost(i,j))
```

### 状压DP（TSP例子）
```python
# 用二进制表示集合，如 dp[mask][u] = 经过mask集合且最后在u的最短路径
dp = [[inf]*n for _ in range(1<<n)]
dp[1][0] = 0
for mask in range(1<<n):
    for u in range(n):
        if not (mask>>u & 1): continue
        for v in range(n):
            if mask>>v & 1: continue
            dp[mask|1<<v][v] = min(dp[mask|1<<v][v], dp[mask][u] + cost[u][v])
```

### 数位DP（记忆化搜索）
```python
@lru_cache(None)
def dfs(pos, tight, ...):
    if pos == len(num): return 1
    limit = int(num[pos]) if tight else 9
    ans = 0
    for d in range(limit+1):
        ans += dfs(pos+1, tight and d==limit, ...)
    return ans
# 调用：dfs(0, True, ...)
```

### 期望DP（逆推）
```python
dp = [0] * (n+1)
for i in range(n-1, -1, -1):
    # 从i可以走到 i+1, i+2, ..., i+k，概率相等
    for step in range(1, k+1):
        dp[i] += (dp[i+step] + 1) / k
```
