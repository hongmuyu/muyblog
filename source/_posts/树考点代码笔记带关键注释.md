## 一、基础定义与通用函数

```python
class TreeNode:
    """二叉树节点"""
    def __init__(self, val=0, left=None, right=None):
        self.val = val          # 节点值
        self.left = left        # 左孩子(TreeNode或None)
        self.right = right      # 右孩子(TreeNode或None)

def inorder(root, res=None):
    """
    中序遍历二叉树，将节点值存入列表。
    参数:
        root: TreeNode, 二叉树的根节点
        res: List, 用于存储遍历结果的列表（可选，默认None时会新建）
    返回值:
        List: 中序遍历结果列表
    """
    if res is None:
        res = []
    if root:
        inorder(root.left, res)
        res.append(root.val)
        inorder(root.right, res)
    return res
```

## 二、二叉树的遍历（前序、中序、后序）

**题目**：给定一个二叉树，输出它的前序、中序、后序遍历结果。

```python
# ---------- 递归 ----------
def preorder_rec(root, res):
    """
    前序遍历（递归）。
    参数:
        root: TreeNode, 当前子树的根节点
        res: List, 存储遍历结果的列表
    返回值: None（结果直接追加到res中）
    """
    if not root: return
    res.append(root.val)
    preorder_rec(root.left, res)
    preorder_rec(root.right, res)

def inorder_rec(root, res):
    """
    中序遍历（递归）。
    参数:
        root: TreeNode, 当前子树的根节点
        res: List, 存储遍历结果的列表
    返回值: None
    """
    if not root: return
    inorder_rec(root.left, res)
    res.append(root.val)
    inorder_rec(root.right, res)

def postorder_rec(root, res):
    """
    后序遍历（递归）。
    参数:
        root: TreeNode, 当前子树的根节点
        res: List, 存储遍历结果的列表
    返回值: None
    """
    if not root: return
    postorder_rec(root.left, res)
    postorder_rec(root.right, res)
    res.append(root.val)

# ---------- 迭代 ----------
def preorder_iter(root):
    """
    前序遍历（迭代，使用栈）。
    参数:
        root: TreeNode, 二叉树的根节点
    返回值:
        List: 前序遍历结果列表
    """
    if not root: return []
    stack, res = [root], []
    while stack:
        node = stack.pop()
        res.append(node.val)
        if node.right: stack.append(node.right)
        if node.left: stack.append(node.left)
    return res

def inorder_iter(root):
    """
    中序遍历（迭代，使用栈+指针）。
    参数:
        root: TreeNode, 二叉树的根节点
    返回值:
        List: 中序遍历结果列表
    """
    stack, res = [], []
    cur = root
    while cur or stack:
        while cur:
            stack.append(cur)
            cur = cur.left
        cur = stack.pop()
        res.append(cur.val)
        cur = cur.right
    return res

def postorder_iter(root):
    """
    后序遍历（迭代，使用双栈技巧：先得到“根右左”，再反转）。
    参数:
        root: TreeNode, 二叉树的根节点
    返回值:
        List: 后序遍历结果列表
    """
    if not root: return []
    stack, res = [root], []
    while stack:
        node = stack.pop()
        res.append(node.val)
        if node.left: stack.append(node.left)
        if node.right: stack.append(node.right)
    return res[::-1]
```

## 三、BST 插入与查找

**题目**：实现二叉查找树的插入和查找操作。

```python
def insert_bst(root, val):
    """
    向BST中插入一个值（忽略重复值）。
    参数:
        root: TreeNode, BST的根节点（可为None）
        val: int/任意可比较类型, 要插入的值
    返回值:
        TreeNode: 插入后BST的根节点
    """
    if not root:
        return TreeNode(val)
    if val < root.val:
        root.left = insert_bst(root.left, val)
    elif val > root.val:
        root.right = insert_bst(root.right, val)
    return root

def search_bst(root, target):
    """
    在BST中查找目标值。
    参数:
        root: TreeNode, BST的根节点
        target: int/任意可比较类型, 要查找的值
    返回值:
        TreeNode: 找到的节点（若存在），否则None
    """
    if not root or root.val == target:
        return root
    if target < root.val:
        return search_bst(root.left, target)
    else:
        return search_bst(root.right, target)
```

## 四、判断是否为合法 BST

**题目**：给定一棵二叉树，判断它是否是合法的二叉搜索树。

```python
def is_bst_inorder(root):
    """
    方法1：通过中序遍历是否严格递增来判断BST合法性。
    参数:
        root: TreeNode, 二叉树的根节点
    返回值:
        bool: True表示是合法BST，False表示不是
    """
    prev = [float('-inf')]
    def inorder(node):
        if not node: return True
        if not inorder(node.left): return False
        if node.val <= prev[0]: return False
        prev[0] = node.val
        return inorder(node.right)
    return inorder(root)

def is_bst_limits(root, min_val=float('-inf'), max_val=float('inf')):
    """
    方法2：递归传递上下界来判断BST合法性。
    参数:
        root: TreeNode, 当前子树的根节点
        min_val: int/float, 当前节点允许的最小值（开区间）
        max_val: int/float, 当前节点允许的最大值（开区间）
    返回值:
        bool: True表示是合法BST，False表示不是
    """
    if not root: return True
    if not (min_val < root.val < max_val): return False
    return (is_bst_limits(root.left, min_val, root.val) and
            is_bst_limits(root.right, root.val, max_val))
```

## 五、根据前序遍历重构 BST

**题目**：给定一个二叉搜索树的前序遍历序列，重构出原二叉树。

```python
def bst_from_preorder(preorder):
    """
    根据BST的前序遍历序列重建BST。
    参数:
        preorder: List[int], BST的前序遍历序列
    返回值:
        TreeNode: 重建后的BST根节点
    """
    idx = 0
    n = len(preorder)
    def build(limit=float('inf')):
        nonlocal idx
        if idx >= n or preorder[idx] > limit:
            return None
        root = TreeNode(preorder[idx])
        idx += 1
        root.left = build(root.val)
        root.right = build(limit)
        return root
    return build()
```

## 六、平衡二叉树（AVL 旋转与插入）

**题目**：实现 AVL 树的插入操作，保持树的高度平衡。

```python
def get_height(node):
    """
    获取节点的高度。
    参数:
        node: TreeNode, 节点（可为None）
    返回值:
        int: 节点的高度（空节点高度0）
    """
    return node.height if node else 0

def update_height(node):
    """
    更新节点的高度（基于左右孩子的高度）。
    参数:
        node: TreeNode, 非空节点
    返回值: None
    """
    node.height = 1 + max(get_height(node.left), get_height(node.right))

def balance_factor(node):
    """
    计算节点的平衡因子（左高 - 右高）。
    参数:
        node: TreeNode, 非空节点
    返回值:
        int: 平衡因子
    """
    return get_height(node.left) - get_height(node.right)

def right_rotate(y):
    """
    右旋操作，用于LL失衡。
    参数:
        y: TreeNode, 失衡的节点
    返回值:
        TreeNode: 旋转后新的子树根节点
    """
    x = y.left
    t2 = x.right
    x.right = y
    y.left = t2
    update_height(y)
    update_height(x)
    return x

def left_rotate(x):
    """
    左旋操作，用于RR失衡。
    参数:
        x: TreeNode, 失衡的节点
    返回值:
        TreeNode: 旋转后新的子树根节点
    """
    y = x.right
    t2 = y.left
    y.left = x
    x.right = t2
    update_height(x)
    update_height(y)
    return y

def insert_avl(root, val):
    """
    向AVL树中插入一个值，并保持平衡。
    参数:
        root: TreeNode, AVL树的根节点
        val: int, 要插入的值
    返回值:
        TreeNode: 插入并平衡后的根节点
    """
    if not root:
        node = TreeNode(val)
        node.height = 1
        return node
    if val < root.val:
        root.left = insert_avl(root.left, val)
    elif val > root.val:
        root.right = insert_avl(root.right, val)
    else:
        return root
    update_height(root)
    bal = balance_factor(root)
    # LL
    if bal > 1 and val < root.left.val:
        return right_rotate(root)
    # RR
    if bal < -1 and val > root.right.val:
        return left_rotate(root)
    # LR
    if bal > 1 and val > root.left.val:
        root.left = left_rotate(root.left)
        return right_rotate(root)
    # RL
    if bal < -1 and val < root.right.val:
        root.right = right_rotate(root.right)
        return left_rotate(root)
    return root
    
# 依次插入以下值
insert_values = [10, 20, 30, 40, 50, 25]
root = None
for v in insert_values:
    root = insert_avl(root, v)
    print(f"插入 {v} 后，树的中序遍历: {inorder(root, [])}   根高度: {root.height}")
# 期望输出（中序遍历为升序，且树始终保持平衡）
# 插入 10 后，树的中序遍历: [10]    根高度: 1
# 插入 20 后，树的中序遍历: [10, 20]    根高度: 2
# 插入 30 后，树的中序遍历: [10, 20, 30]    根高度: 2   （触发左旋，新根20）
# 插入 40 后，树的中序遍历: [10, 20, 30, 40]    根高度: 3
# 插入 50 后，树的中序遍历: [10, 20, 30, 40, 50]    根高度: 3   （触发左旋，新根30）
# 插入 25 后，树的中序遍历: [10, 20, 25, 30, 40, 50]    根高度: 3   （触发RL旋转，最终平衡）
# 最终树的结构（平衡因子都满足 |bf|≤1）
print("\n最终树的中序遍历:", inorder(root, []))   # [10,20,25,30,40,50]
print("最终树的根高度:", root.height)             # 3
```

```python
from collections import deque

def is_complete_bst(root):
    """判断一棵树是否为完全二叉树"""
    if not root: return True
    queue = deque([root])
    seen_null = False
    while queue:
        node = queue.popleft()
        if node is None:
            seen_null = True
        else:
            if seen_null:
                return False
            queue.append(node.left)
            queue.append(node.right)
    return True

def build_bst_from_list(nums):
    """根据列表值顺序插入构建BST"""
    root = None
    for v in nums:
        root = insert_bst(root, v)   # 复用之前的insert_bst
    return root
```

## 八、题目2：黑白树 II（插入时维护最大黑白平衡子树）

**题目**：BST 的每个节点有黑白色，定义黑白平衡树为：任意节点的左右子树黑色节点数之差 ≤ 1，且所有子树都平衡。每次插入新节点（含颜色），求当前树中最大的黑白平衡子树的大小。

```python
class ColorNode:
    def __init__(self, val, color):
        self.val = val
        self.left = None
        self.right = None
        self.color = color          # 0:白, 1:黑
        self.black_cnt = 1 if color == 1 else 0
        self.balanced = True

def update_node(node):
    if not node: return
    left_black = node.left.black_cnt if node.left else 0
    right_black = node.right.black_cnt if node.right else 0
    node.black_cnt = (1 if node.color == 1 else 0) + left_black + right_black
    left_bal = node.left.balanced if node.left else True
    right_bal = node.right.balanced if node.right else True
    node.balanced = left_bal and right_bal and abs(left_black - right_black) <= 1

def insert_color_bst(root, val, color):
    if not root:
        return ColorNode(val, color)
    if val < root.val:
        root.left = insert_color_bst(root.left, val, color)
    elif val > root.val:
        root.right = insert_color_bst(root.right, val, color)
    else:
        return root
    update_node(root)
    return root

def get_max_balanced_subtree_size(root):
    if not root:
        return 0
    size = 1
    if root.left:
        size += get_max_balanced_subtree_size(root.left)
    if root.right:
        size += get_max_balanced_subtree_size(root.right)
    if root.balanced:
        return size
    else:
        return max(get_max_balanced_subtree_size(root.left),
                   get_max_balanced_subtree_size(root.right))

# 输入与输出示例
insertions = [(5,1), (3,0), (8,1), (1,1), (4,0)]   # (值,颜色) 1黑0白
root = None
for val, col in insertions:
    root = insert_color_bst(root, val, col)
    size = get_max_balanced_subtree_size(root)
    print(f"插入 ({val},{'黑' if col else '白'}) 后，最大平衡子树大小 = {size}")

# 输出:
# 插入 (5,黑) 后，最大平衡子树大小 = 1
# 插入 (3,白) 后，最大平衡子树大小 = 2
# 插入 (8,黑) 后，最大平衡子树大小 = 3
# 插入 (1,黑) 后，最大平衡子树大小 = 4
# 插入 (4,白) 后，最大平衡子树大小 = 5
```

## 九、题目3：重复元素序列插入 BST，求查找总次数

**题目**：给定一个含重复元素的序列，依次插入空BST（重复元素不创建新节点，但统计次数），求查找所有元素所需的总比较次数。

```python
from collections import Counter

class NodeWithDepth(TreeNode):
    def __init__(self, val=0, left=None, right=None, depth=0):
        super().__init__(val, left, right)
        self.depth = depth

def insert_and_record_depth(root, val, depth, record):
    if root is None:
        node = NodeWithDepth(val, depth=depth)
        record[val] = depth
        return node
    if val < root.val:
        root.left = insert_and_record_depth(root.left, val, depth+1, record)
    elif val > root.val:
        root.right = insert_and_record_depth(root.right, val, depth+1, record)
    else:
        record[val] = root.depth
    return root

def total_search_costs(nums):
    freq = Counter(nums)
    root = None
    depth_map = {}
    for val in freq.keys():
        root = insert_and_record_depth(root, val, 0, depth_map)
    total = 0
    for val, cnt in freq.items():
        total += cnt * (depth_map[val] + 1)
    return total

# 输入与输出示例
nums1 = [5, 3, 8, 3, 5, 7]
print(f"输入序列: {nums1}")
print(f"总查找次数: {total_search_costs(nums1)}")   # 输出: 11

nums2 = [2, 1, 3, 2, 2, 4, 1, 5]
print(f"输入序列: {nums2}")
print(f"总查找次数: {total_search_costs(nums2)}")   # 输出: 20 (实际运行结果)
```

### 1. 二叉树的层序遍历
**题目**：给定一棵二叉树，返回其层序遍历的结果（每一层一个列表）。

```python
def level_order(root):
    """
    层序遍历二叉树（BFS）。
    参数:
        root: TreeNode, 二叉树的根节点
    返回值:
        List[List[int]]: 按层组织的节点值列表
    """
    if not root: return []
    queue = deque([root])
    res = []
    while queue:
        level = []
        for _ in range(len(queue)):
            node = queue.popleft()
            level.append(node.val)
            if node.left: queue.append(node.left)
            if node.right: queue.append(node.right)
        res.append(level)
    return res
```

### 2. 二叉树的最大深度
**题目**：求二叉树的最大深度。

```python
def max_depth(root):
    """

    参数:
        root: TreeNode, 二叉树的根节点
```

### 3. 二叉树的直径
**题目**：求二叉树的直径（任意两节点间最长路径的边数）。

```python
def diameter_of_binary_tree(root):
    """
    计算二叉树的直径（边数）。
    参数:
        root: TreeNode, 二叉树的根节点
    返回值:
        int: 直径长度（边数）
    """
    diameter = 0
    def depth(node):
        nonlocal diameter
        if not node: return 0
        left = depth(node.left)
        right = depth(node.right)
        diameter = max(diameter, left + right)
        return 1 + max(left, right)
    depth(root)
    return diameter
```

### 4. 最低公共祖先（LCA）
**题目**：给定二叉树和两个节点，求它们的最低公共祖先。

```python
def lowest_common_ancestor(root, p, q):
    """
    求二叉树中两个节点的最低公共祖先。
    参数:
        root: TreeNode, 二叉树的根节点
        p: TreeNode, 节点p
        q: TreeNode, 节点q
    返回值:
        TreeNode: p和q的LCA节点
    """
    if not root or root == p or root == q:
        return root
    left = lowest_common_ancestor(root.left, p, q)
    right = lowest_common_ancestor(root.right, p, q)
    if left and right:
        return root
    return left if left else right
```

### 5. BST中第K小的元素
**题目**：给定一棵二叉搜索树，找出其中第 k 小的元素（k 从1开始）。

```python
def kth_smallest_bst(root, k):
    """
    在BST中查找第k小的元素（中序遍历迭代版）。
    参数:
        root: TreeNode, BST的根节点
        k: int, 排名（1-indexed）
    返回值:
        int: 第k小的节点值，若k超出范围则返回-1
    """
    stack = []
    cur = root
    while cur or stack:
        while cur:
            stack.append(cur)
            cur = cur.left
        cur = stack.pop()
        k -= 1
        if k == 0:
            return cur.val
        cur = cur.right
    return -1
```

### 7. 二叉树的序列化与反序列化
**题目**：设计一个算法，将二叉树序列化为字符串，并能反序列化回来。

```python
def serialize(root):
    """
    将二叉树序列化为字符串（前序遍历 + "null"标记空节点）。
    参数:
        root: TreeNode, 二叉树的根节点
    返回值:
        str: 序列化后的字符串
    """
    res = []
    def dfs(node):
        if not node:
            res.append("null")
            return
        res.append(str(node.val))
        dfs(node.left)
        dfs(node.right)
    dfs(root)
    return ",".join(res)

def deserialize(data):
    """
    将序列化字符串反序列化为二叉树。
    参数:
        data: str, serialize生成的字符串
    返回值:
        TreeNode: 重建后的二叉树根节点
    """
    vals = data.split(",")
    idx = 0
    def build():
        nonlocal idx
        if vals[idx] == "null":
            idx += 1
            return None
        node = TreeNode(int(vals[idx]))
        idx += 1
        node.left = build()
        node.right = build()
        return node
    return build()
```

## 十一、BST 删除操作

**题目**：实现二叉查找树的删除操作，删除指定值的节点，并保持 BST 性质。

```python
def delete_bst(root, key):
    """
    从BST中删除值为key的节点。
    参数:
        root: TreeNode, BST的根节点
        key: int, 要删除的值
    返回值:
        TreeNode: 删除后的BST根节点
    """
    if not root:
        return None

    # 1. 查找要删除的节点
    if key < root.val:
        root.left = delete_bst(root.left, key)
    elif key > root.val:
        root.right = delete_bst(root.right, key)
    else:
        # 找到要删除的节点
        # 情况1：叶子节点（无子节点）
        if not root.left and not root.right:
            return None
        # 情况2：只有一个子节点
        if not root.left:
            return root.right
        if not root.right:
            return root.left
        # 情况3：有两个子节点
        # 关键：找到右子树的最小节点（或左子树的最大节点）
        successor = find_min(root.right)   # 右子树最左节点
        root.val = successor.val           # 替换值
        # 关键：删除右子树中的那个最小节点
        root.right = delete_bst(root.right, successor.val)

    return root

def find_min(node):
    """
    找到以node为根的树中的最小值节点。
    参数:
        node: TreeNode, 非空节点
    返回值:
        TreeNode: 最小值节点
    """
    while node.left:
        node = node.left
    return node

def find_max(node):
    """
    找到以node为根的树中的最大值节点（备用，也可用于删除）。
    参数:
        node: TreeNode, 非空节点
    返回值:
        TreeNode: 最大值节点
    """
    while node.right:
        node = node.right
    return node
```

**补充说明**：也可以用左子树的最大节点来替换，只需将`successor` 改为`find_max(root.left)` 即可。

## 十二、AVL 树的删除（扩展，可选）

**题目**：实现 AVL 树的删除操作，删除后保持平衡。

```python
def delete_avl(root, key):
    """
    从AVL树中删除值为key的节点，并保持平衡。
    参数:
        root: TreeNode, AVL树的根节点（需有height属性）
        key: int, 要删除的值
    返回值:
        TreeNode: 删除并平衡后的根节点
    """
    if not root:
        return None

    # 1. 标准BST删除
    if key < root.val:
        root.left = delete_avl(root.left, key)
    elif key > root.val:
        root.right = delete_avl(root.right, key)
    else:
        # 找到要删除的节点
        if not root.left or not root.right:
            # 0或1个子节点
            child = root.left if root.left else root.right
            return child
        else:
            # 两个子节点：用后继替换
            successor = find_min(root.right)
            root.val = successor.val
            root.right = delete_avl(root.right, successor.val)

    if not root:
        return root

    # 2. 更新高度并平衡
    update_height(root)
    bal = balance_factor(root)

    # 左子树失衡（左高 > 右高+1）
    if bal > 1:
        # 左-左 或 左-右
        if balance_factor(root.left) >= 0:   # LL
            return right_rotate(root)
        else:                                # LR
            root.left = left_rotate(root.left)
            return right_rotate(root)
    # 右子树失衡
    if bal < -1:
        if balance_factor(root.right) <= 0:  # RR
            return left_rotate(root)
        else:                                # RL
            root.right = right_rotate(root.right)
            return left_rotate(root)

    return root
```

**注意**：AVL 删除涉及四种失衡情况与插入类似，但需要从删除节点向上回溯平衡，这里用递归自然实现了回溯。
